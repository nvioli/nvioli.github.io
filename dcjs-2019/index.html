var poly = new Tone.PolySynth(4, Tone.Synth).toMaster();
var fmSynth = new Tone.FMSynth({volume:-10}).toMaster();
// TODO set modulationIndex

var drumCompress = new Tone.Compressor({ "threshold" : -30, "ratio" : 6, "attack" : 0.3, "release" : 0.1}).toMaster();
var distortion = new Tone.Distortion({ "distortion" : 0.4, "wet" : 0.4});
var kick = new Tone.MembraneSynth({ "pitchDecay" : 0.01, "octaves" : 6, "oscillator" : { "type" : "square4" }, "envelope" : { "attack" : 0.001, "decay" : 0.2, "sustain" : 0 }}).connect(drumCompress);
var snare = new Tone.Player({ "url" : "snare.mp3", "retrigger" : true, "fadeOut" : 0.1, "volume": -20}).chain(distortion, drumCompress);

var bass = new Tone.Loop(time => {
	//poly.triggerAttackRelease("F1", "8n", time);
	//poly.triggerAttackRelease(["F2","Ab2","C3"], "8n", "+8n");
}, "4n").start(0);

var drum1 = new Tone.Loop(time => {
   kick.triggerAttackRelease("C2", "8n",time);
}, "1n").start(0);

var drum2 = new Tone.Loop(time => {
   kick.triggerAttackRelease("G2", "8n",time);
}, "4n").start("16n");

var drum3 = new Tone.Loop(time => {
   snare.start(time).stop(time + 0.1);
}, "4n").start("8n");

var musicForAirportsNotes = ["F2","Ab2","C3","Db3","Eb3","F3","Ab3"];
var mfa = new Tone.Loop(time => {
   fmSynth.triggerAttackRelease(musicForAirportsNotes[Math.floor(Math.random()*musicForAirportsNotes.length)],"4n",time);
}, "4n").start("2n");

drum2.probability = 0.5;
drum3.probability = 0.25;

Tone.Transport.start();
